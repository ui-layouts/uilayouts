
## Usage

```tsx
import { KineticImages } from '@/components/ui/image-effect-kinetic-images';

export default function Example() {
  return <KineticImages images={['/photos/first.jpg', '/photos/second.jpg']} />;
}
```

Drag or scroll within the gallery to explore. Focus the canvas and use the arrow keys for keyboard navigation.
