```model.tsx 
import { createPortal, useFrame, useLoader, useThree } from '@react-three/fiber';
import { useEffect, useMemo, useRef } from 'react';
import * as THREE from 'three';
import { vertex, fragment } from './shaders';

const waveIndices = Array.from({ length: 100 }, (_, index) => index);
const imageUrls = ['/picture1.jpeg', '/picture2.jpeg', '/picture3.jpeg'];
type WaveMesh = THREE.Mesh<THREE.PlaneGeometry, THREE.MeshBasicMaterial>;

export default function Model() {
  const { width, height } = useThree((state) => state.size);
  const brush = useLoader(THREE.TextureLoader, '/brush.png');
  const [first, second, third] = useLoader(THREE.TextureLoader, imageUrls);
  const waves = useRef<Array<WaveMesh | null>>([]);
  const previousPointer = useRef(new THREE.Vector2());
  const nextWave = useRef(0);

  const waveScene = useMemo(() => new THREE.Scene(), []);
  const imageGeometry = useMemo(() => new THREE.PlaneGeometry(1, 1), []);
  const imageMaterials = useMemo(
    () => [first, second, third].map((map) => new THREE.MeshBasicMaterial({ map })),
    [first, second, third]
  );
  const imageScene = useMemo(() => {
    const scene = new THREE.Scene();
    imageMaterials.forEach((material, index) => {
      const mesh = new THREE.Mesh(imageGeometry, material);
      mesh.position.set(((index - 1) * width) / 4, 0, 1);
      mesh.scale.set(width / 5, width / 4, 1);
      scene.add(mesh);
    });
    return scene;
  }, [imageGeometry, imageMaterials, width]);
  const imageCamera = useMemo(() => {
    const camera = new THREE.OrthographicCamera(
      -width / 2, width / 2, height / 2, -height / 2, 0.1, 100
    );
    camera.position.z = 2;
    return camera;
  }, [width, height]);

  const waveBuffer = useMemo(() => new THREE.WebGLRenderTarget(1, 1), []);
  const imageBuffer = useMemo(() => new THREE.WebGLRenderTarget(1, 1), []);
  const uniforms = useMemo(
    () => ({
      uDisplacement: { value: waveBuffer.texture },
      uTexture: { value: imageBuffer.texture },
      winResolution: { value: new THREE.Vector2() },
    }),
    [imageBuffer, waveBuffer]
  );

  useEffect(() => () => imageGeometry.dispose(), [imageGeometry]);
  useEffect(() => () => {
    // The loader caches the shared textures. Only dispose the materials we created.
    imageMaterials.forEach((material) => material.dispose());
  }, [imageMaterials]);
  useEffect(() => () => {
    waveBuffer.dispose();
    imageBuffer.dispose();
  }, [imageBuffer, waveBuffer]);

  useFrame(({ gl, scene, camera, pointer }) => {
    const x = pointer.x * width / 2;
    const y = pointer.y * height / 2;
    if (Math.abs(x - previousPointer.current.x) > 0.1 || Math.abs(y - previousPointer.current.y) > 0.1) {
      const wave = waves.current[nextWave.current];
      if (wave) {
        wave.position.set(x, y, 0);
        wave.visible = true;
        wave.material.opacity = 1;
        wave.scale.set(1.75, 1.75, 1);
      }
      nextWave.current = (nextWave.current + 1) % waveIndices.length;
    }
    previousPointer.current.set(x, y);

    for (const wave of waves.current) {
      if (!wave?.visible) continue;
      wave.rotation.z += 0.025;
      wave.material.opacity *= 0.95;
      wave.scale.x = 0.98 * wave.scale.x + 0.155;
      wave.scale.y = 0.98 * wave.scale.y + 0.155;
      if (wave.material.opacity < 0.01) wave.visible = false;
    }

    // Measure the resized drawing buffer before rendering any of this frame's passes.
    const resolution = gl.getDrawingBufferSize(uniforms.winResolution.value);
    if (waveBuffer.width !== resolution.x || waveBuffer.height !== resolution.y) {
      waveBuffer.setSize(resolution.x, resolution.y);
      imageBuffer.setSize(resolution.x, resolution.y);
    }

    gl.setRenderTarget(waveBuffer);
    gl.clear();
    gl.render(waveScene, camera);
    gl.setRenderTarget(imageBuffer);
    gl.clear();
    gl.render(imageScene, imageCamera);
    gl.setRenderTarget(null);
    gl.render(scene, camera);
  }, 1);

  return (
    <>
      {createPortal(
        <group>
          {waveIndices.map((index) => (
            <mesh key={index} visible={false} ref={(mesh) => {
              waves.current[index] = mesh as WaveMesh | null;
            }}>
              <planeGeometry args={[60, 60]} />
              <meshBasicMaterial transparent map={brush} />
            </mesh>
          ))}
        </group>,
        waveScene
      )}
      <mesh>
        <planeGeometry args={[width, height]} />
        <shaderMaterial vertexShader={vertex} fragmentShader={fragment} uniforms={uniforms} />
      </mesh>
    </>
  );
}
```

```shaders.tsx 
export const fragment = `
uniform sampler2D uTexture;
uniform sampler2D uDisplacement;
uniform vec2 winResolution;
varying vec2 vUv;
float PI = 3.141592653589793238;

void main() {
  vec2 vUvScreen = gl_FragCoord.xy / winResolution;

  vec4 displacement = texture2D(uDisplacement, vUvScreen);
  float theta = displacement.r*2.0*PI;

  vec2 dir = vec2(sin(theta),cos(theta));
  vec2 uv = vUvScreen + dir*displacement.r*0.075;
  vec4 color = texture2D(uTexture,uv);

  gl_FragColor = color;
}
`;

export const vertex = `
varying vec2 vUv;

void main() {
  vUv = uv;
  gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
}
`;
```

## Map the pointer into image coordinates

Normalize pointer positions relative to the canvas. Account for the image crop so the ripple originates where the user actually interacted.

## Animate displacement in a shader

A ripple can be modeled as a traveling sine wave whose strength decays with distance and time. Use a texture sampling offset rather than modifying the original source image.

## Fallback to the original image

Do not require WebGL for content access. Keep a still image for reduced motion or failed context creation, and release resources when the effect unmounts.

## Implementation pattern

The following focused snippet demonstrates the principle, not the complete UI Layouts implementation. The live example and linked documentation contain the full source.

```glsl
float wave = sin(distance(uv, center) * 32.0 - time * 5.0);
float strength = wave * exp(-distance(uv, center) * 5.0);
```
